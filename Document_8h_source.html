379template <typename T>
380[[nodiscard]]
381std::optional<Cmper> importTextInto(const DocumentPtr& target,
382 const MusxInstance<T>& source, const ImportObjectCallback& onImported)
383{
384 static_assert(is_pool_type_v<TextsPool, T>, "Type T is not registered in TextsPool");
385 MUSX_ASSERT_IF(!target) {
386 throw std::invalid_argument("importTextInto received a null target document");
387 }
388 MUSX_ASSERT_IF(!source) {
389 throw std::invalid_argument("importTextInto received null text");
390 }
391
392 std::optional<Cmper> importedId;
393 if constexpr (std::is_same_v<T, FileInfoText>) {
394 importedId = source->getTextNumber();
395 if (target->getTexts()->get<FileInfoText>(*importedId)) {
396 return std::nullopt;
397 }
398 } else {
399 importedId = target->getTexts()->nextFreeCmper<T>();
400 }
401 if (!importedId) {
402 return std::nullopt;
403 }
404 auto imported = std::make_shared<T>(
405 target, SCORE_PARTID, EnigmaBase::ShareMode::All, *importedId);
406 *imported = *source;
407 target->getTexts()->add(T::XmlNodeName, imported);
408 if (onImported) {
409 onImported(*imported);
410 }
411 if constexpr (std::is_base_of_v<LyricsTextBase, T>) {
412 imported->createSyllableInfo(imported);
413 }
414 return importedId;
415}

417template <typename T>
418[[nodiscard]]
419std::optional<Cmper> importTextInto(const DocumentPtr& target,
420 const MusxInstance<T>& source)
421{
422 return importTextInto(target, source, {});
423}
