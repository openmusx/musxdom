132class PlaybackResourcesFactory

134public:
147 template <typename XmlDocumentType>
148 [[nodiscard]] static dom::PlaybackResourcesPtr create(const PlaybackResourceInputs& inputs)
149 {
150 static_assert(std::is_base_of_v<xml::IXmlDocument, XmlDocumentType>,
151 "XmlDocumentType must derive from IXmlDocument.");
152 const auto parse = [](const PlaybackResourceFile& file, const char* rootName) -> std::unique_ptr<xml::IXmlDocument> {
153 if (!file.data || file.size == 0) {
154 return nullptr;
155 }
156 auto xmlDocument = std::make_unique<XmlDocumentType>();
157 try {
158 xmlDocument->loadFromBuffer(file.data, file.size);
159 } catch (const std::exception& ex) {
160 throw std::invalid_argument("Unable to parse playback resource file \"" + file.fileName + "\": " + ex.what());
161 }
162 validateRoot(*xmlDocument, file, rootName);
163 return xmlDocument;
164 };
165 ParsedFiles soundmaps;
166 for (const auto& file : inputs.soundmaps) {
167 if (auto xmlDocument = parse(file, SOUNDMAP_ROOT)) {
168 soundmaps.emplace_back(file.fileName, std::move(xmlDocument));
169 }
170 }
171 ParsedFiles deviceAnnotations;
172 for (const auto& file : inputs.deviceAnnotations) {
173 if (auto xmlDocument = parse(file, DEVICE_ANNOTATION_ROOT)) {
174 deviceAnnotations.emplace_back(file.fileName, std::move(xmlDocument));
175 }
176 }
177 ParsedFiles instrumentCatalogs;
178 for (const auto& file : inputs.instrumentCatalogs) {
179 if (auto xmlDocument = parse(file, INSTRUMENT_CATALOG_ROOT)) {
180 instrumentCatalogs.emplace_back(file.fileName, std::move(xmlDocument));
181 }
182 }
183 return createFromXml(soundmaps, deviceAnnotations, instrumentCatalogs);
184 }
185
186private:
187 static constexpr const char* SOUNDMAP_ROOT = "Soundmap";
188 static constexpr const char* DEVICE_ANNOTATION_ROOT = "FinaleNameDocument";
189 static constexpr const char* INSTRUMENT_CATALOG_ROOT = "InstrumentCatalog";
190
191 using ParsedFiles = std::vector<std::pair<std::string, std::unique_ptr<xml::IXmlDocument>>>;
192
193 static void validateRoot(const xml::IXmlDocument& xmlDocument, const PlaybackResourceFile& file, const char* rootName);
194 static dom::PlaybackResourcesPtr createFromXml(const ParsedFiles& soundmaps, const ParsedFiles& deviceAnnotations,
195 const ParsedFiles& instrumentCatalogs);
196};
