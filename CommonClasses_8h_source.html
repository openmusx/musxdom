711class TimeSignature : public DocumentElementNoPart
712{
713public:
716 enum class Abbreviation
717 {
718 NotApplicable,
719 Numeric,
720 Abbreviated
721 };
722
724 struct TimeSigComponent
725 {
726 std::vector<util::Fraction> counts;
728 std::vector<Edu> units;
731
733 bool operator==(const TimeSigComponent& src) const
734 { return counts == src.counts && units == src.units; }
735
737 util::Fraction sumCounts() const
738 { return std::accumulate(counts.begin(), counts.end(), util::Fraction{}); }
739
741 Edu sumUnits() const
742 { return std::accumulate(units.begin(), units.end(), Edu{}); }
743
746 static std::pair<util::Fraction, Edu> normalizeCompoundUnit(util::Fraction count, Edu unit);
747
751 TimeSigComponent normalizeCompound() const;
752 };
753
754 std::vector<TimeSigComponent> components;
755
760 std::pair<util::Fraction, NoteType> calcSimplified() const;
761
763 util::Fraction calcTotalDuration() const
764 {
765 util::Fraction result = std::accumulate(components.begin(), components.end(), util::Fraction{},
766 [](const util::Fraction& acc, const TimeSigComponent& comp)
767 { return acc + (comp.sumCounts() * comp.sumUnits()); }
768 );
769 return result / Edu(NoteType::Whole);
770 }
771
781 util::Fraction calcBeatValueAt(Edu eduPosition) const;
782
784 bool isSame(const TimeSignature& src) const;
785
789 MusxInstance<TimeSignature> createComponent(size_t index) const
790 {
791 checkIndex(index);
792 return MusxInstance<TimeSignature>(new TimeSignature(getDocument(), components[index], m_abbreviation));
793 }
794
799 std::optional<char32_t> getAbbreviatedSymbol() const;
800
802 bool isCommonTime() const;
804 bool isCutTime() const;
805
806private:
807 void checkIndex(size_t index) const
808 {
809 if (index > components.size()) {
810 throw std::invalid_argument("Index out of range. The time signature has " + std::to_string(components.size())
811 + " elements. The index requested was " + std::to_string(index) + ".");
812 }
813 }
814
816 explicit TimeSignature(const DocumentWeakPtr& document, int beats, Edu unit, bool hasCompositeTop, bool hasCompositeBottom,
817 Abbreviation abbreviate = {});
818
820 explicit TimeSignature(const DocumentWeakPtr& document, const TimeSigComponent& timeSigUnit, Abbreviation abbreviate = {})
821 : DocumentElementNoPart(document), m_abbreviation(abbreviate)
822 {
823 components.push_back(timeSigUnit);
824 }
825
826 Abbreviation m_abbreviation{};
827

829 friend class others::OssiaHeader;
830 friend class details::IndependentStaffDetails;
831};
