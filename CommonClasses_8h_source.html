705class TimeSignature : public DocumentElementNoPart
706{
707public:
710 enum class Abbreviation
711 {
712 NotApplicable,
713 Numeric,
714 Abbreviated
715 };
716
718 struct TimeSigComponent
719 {
720 std::vector<util::Fraction> counts;
722 std::vector<Edu> units;
725
727 bool operator==(const TimeSigComponent& src) const
728 { return counts == src.counts && units == src.units; }
729
731 util::Fraction sumCounts() const
732 { return std::accumulate(counts.begin(), counts.end(), util::Fraction{}); }
733
735 Edu sumUnits() const
736 { return std::accumulate(units.begin(), units.end(), Edu{}); }
737
740 static std::pair<util::Fraction, Edu> normalizeCompoundUnit(util::Fraction count, Edu unit);
741
745 TimeSigComponent normalizeCompound() const;
746 };
747
748 std::vector<TimeSigComponent> components;
749
754 std::pair<util::Fraction, NoteType> calcSimplified() const;
755
757 util::Fraction calcTotalDuration() const
758 {
759 util::Fraction result = std::accumulate(components.begin(), components.end(), util::Fraction{},
760 [](const util::Fraction& acc, const TimeSigComponent& comp)
761 { return acc + (comp.sumCounts() * comp.sumUnits()); }
762 );
763 return result / Edu(NoteType::Whole);
764 }
765
775 util::Fraction calcBeatValueAt(Edu eduPosition) const;
776
778 bool isSame(const TimeSignature& src) const;
779
783 MusxInstance<TimeSignature> createComponent(size_t index) const
784 {
785 checkIndex(index);
786 return MusxInstance<TimeSignature>(new TimeSignature(getDocument(), components[index], m_abbreviation));
787 }
788
793 std::optional<char32_t> getAbbreviatedSymbol() const;
794
796 bool isCommonTime() const;
798 bool isCutTime() const;
799
800private:
801 void checkIndex(size_t index) const
802 {
803 if (index > components.size()) {
804 throw std::invalid_argument("Index out of range. The time signature has " + std::to_string(components.size())
805 + " elements. The index requested was " + std::to_string(index) + ".");
806 }
807 }
808
810 explicit TimeSignature(const DocumentWeakPtr& document, int beats, Edu unit, bool hasCompositeTop, bool hasCompositeBottom,
811 Abbreviation abbreviate = {});
812
814 explicit TimeSignature(const DocumentWeakPtr& document, const TimeSigComponent& timeSigUnit, Abbreviation abbreviate = {})
815 : DocumentElementNoPart(document), m_abbreviation(abbreviate)
816 {
817 components.push_back(timeSigUnit);
818 }
819
820 Abbreviation m_abbreviation{};
821

823 friend class others::OssiaHeader;
824 friend class details::IndependentStaffDetails;
825};
